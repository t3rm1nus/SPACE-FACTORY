# Tests de continuación adaptativa (máx. 3 intentos: inicial + 2 continuaciones)

def _fake_result(text: str) -> Any:
    class _Result:
        text = text
        raw_response = {"model": "qwen-agent:latest", "context": {}}

        output_tokens = 20

    long_md = "word " * 2000

def test_continuation_three_attempts_still_fails(monkeypatch: pytest.MonkeyPatch) -> None:
    """3 intentos (inicial + 2 cont.) siguen por debajo -> FAIL y quality gate no se falsea."""
    import modules.chapter_writer.main as main

    short_md = "word " * 100
    responses = [_fake_result(short_md)] * 3

    class FakeProvider:
        name = "ollama"
        model = "qwen-agent:latest"

        def __init__(self) -> None:
            self.calls = 0

        def generate(self, *args: Any, **kwargs: Any) -> Any:
            idx = min(self.calls, len(responses) - 1)
            self.calls += 1
            return responses[idx]

    provider = FakeProvider()
    monkeypatch.setattr(main, "get_provider", lambda: provider)
    monkeypatch.setattr(main, "DEFAULT_ROUTER_MODEL", "qwen-agent:latest")
    monkeypatch.setattr(main, "_write_artifacts", lambda *args, **kwargs: "data/artifacts/1/chapter_1/chapter.md")

    out = execute(_payload())
    assert out["word_count"] < 1500
    assert out["quality_gate"] == "FAIL"
    assert any("menos de 1500 palabras" in str(e) for e in out.get("quality_errors", []))
    assert provider.calls == 3

def test_continuation_target_words_calculation(monkeypatch: pytest.MonkeyPatch) -> None:
    """El prompt de continuación solicita ~remaining_words * 1.3, con mínimo 120."""
    import modules.chapter_writer.main as main

    short_md = "word " * 100
    long_md = "word " * 2000
    captured_prompts = []

    class FakeResult:
        def __init__(self, text: str) -> None:
            self.text = text
            self.raw_response = {"model": "qwen-agent:latest", "context": {}}
            self.input_tokens = 10
            self.output_tokens = 20
            self.provider = "ollama"
            self.model = "qwen-agent:latest"
            self.cost = 0.0

    class FakeProvider:
        name = "ollama"
        model = "qwen-agent:latest"

        def generate(self, *args: Any, **kwargs: Any) -> Any:
            captured_prompts.append(args[0] if args else "")
            if len(captured_prompts) == 1:
                return FakeResult(short_md)
            return FakeResult(long_md)

    monkeypatch.setattr(main, "get_provider", lambda: FakeProvider())
    monkeypatch.setattr(main, "DEFAULT_ROUTER_MODEL", "qwen-agent:latest")
    monkeypatch.setattr(main, "_write_artifacts", lambda *args, **kwargs: "data/artifacts/1/chapter_1/chapter.md")

    out = execute(_payload())
    assert out["word_count"] >= 1500
    assert len(captured_prompts) == 2
    continuation_prompt = captured_prompts[1]
    expected_target = max(int((1500 - 100) * 1.3), 120)
    assert f"approximately {expected_target} additional words" in continuation_prompt
